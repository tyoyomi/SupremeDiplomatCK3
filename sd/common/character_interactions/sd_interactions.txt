### Grant / revoke (when rule allows targets)
sd_grant_supreme_diplomat_interaction = {
	category = interaction_category_friendly
	icon = diplomacy
	interface_priority = 58
	common_interaction = yes

	desc = sd_grant_supreme_diplomat_interaction_desc
	auto_accept = yes

	is_shown = {
		has_game_rule = sd_rule_granting_player_and_targets
		scope:actor = {
			is_alive = yes
			sd_actor_may_bestow_by_rule = yes
		}
		scope:recipient = {
			is_alive = yes
			is_adult = yes
			NOT = { has_trait = sd_supreme_diplomat }
			OR = {
				is_child_of = scope:actor
				is_grandchild_of = scope:actor
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			custom_tooltip = {
				text = sd_tt_can_receive_diplomat
				is_adult = yes
				NOT = { has_trait = sd_supreme_diplomat }
			}
		}
	}

	on_accept = {
		scope:recipient = {
			sd_embrace_supreme_diplomat_effect = yes
		}
	}

	ai_will_do = { base = 0 }
}

### 1 — Offer Vassalage
sd_offer_vassalage_interaction = {
	category = interaction_category_diplomacy
	icon = icon_vassal
	interface_priority = 90
	common_interaction = yes

	desc = sd_offer_vassalage_interaction_desc

	auto_accept = {
		custom_description = {
			text = sd_tt_high_auto_accept
			subject = scope:actor
			object = scope:recipient
			scope:actor = { sd_intensity_is_high = yes }
		}
	}

	is_shown = {
		sd_can_offer_vassalage = yes
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			custom_tooltip = {
				text = sd_tt_can_offer_vassalage
				sd_is_supreme_diplomat = yes
				sd_module_vassalage_on = yes
				NOT = { has_character_flag = sd_vassalage_cd }
				gold >= sd_vassalize_gold_cost_value
			}
		}
		scope:recipient = {
			custom_tooltip = {
				text = sd_tt_independent_valid
				sd_is_independent_valid_target = yes
			}
		}
	}

	cost = {
		gold = {
			value = sd_vassalize_gold_cost_value
		}
		prestige = {
			value = sd_vassalize_prestige_cost_value
		}
	}

	on_accept = {
		sd_offer_vassalage_effect = yes
	}

	ai_accept = {
		base = 20
		modifier = {
			add = 40
			scope:actor = { sd_intensity_is_mid = yes }
		}
		modifier = {
			add = 1000
			scope:actor = { sd_intensity_is_high = yes }
		}
		modifier = {
			add = 30
			opinion = {
				target = scope:actor
				value >= 50
			}
		}
		modifier = {
			add = -40
			scope:actor.primary_title.tier = scope:recipient.primary_title.tier
			scope:actor = { sd_intensity_is_low = yes }
		}
	}

	ai_will_do = { base = 0 }
}
