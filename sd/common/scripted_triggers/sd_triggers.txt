# Supreme Diplomat — scripted triggers

sd_is_supreme_diplomat = {
	has_trait = sd_supreme_diplomat
}

sd_granting_disabled = {
	has_game_rule = sd_rule_granting_disabled
}

sd_can_embrace_diplomat = {
	is_alive = yes
	is_adult = yes
	NOT = { has_game_rule = sd_rule_granting_disabled }
	NOT = { has_trait = sd_supreme_diplomat }
	trigger_if = {
		limit = { has_game_rule = sd_rule_granting_player_self }
		is_ai = no
	}
	trigger_else_if = {
		limit = { has_game_rule = sd_rule_granting_player_and_targets }
		OR = {
			is_ai = no
			has_game_rule = sd_rule_ai_follow_intensity
		}
	}
	trigger_else = {
		always = no
	}
}

sd_actor_may_bestow_by_rule = {
	NOT = { has_game_rule = sd_rule_granting_disabled }
	has_game_rule = sd_rule_granting_player_and_targets
	trigger_if = {
		limit = { is_ai = yes }
		has_game_rule = sd_rule_ai_follow_intensity
	}
}

sd_ai_may_use_diplomat = {
	OR = {
		is_ai = no
		has_game_rule = sd_rule_ai_follow_intensity
	}
}

sd_intensity_is_low = {
	exists = var:sd_cfg_intensity
	var:sd_cfg_intensity = 1
}

sd_intensity_is_mid = {
	exists = var:sd_cfg_intensity
	var:sd_cfg_intensity = 2
}

sd_intensity_is_high = {
	OR = {
		NOT = { exists = var:sd_cfg_intensity }
		var:sd_cfg_intensity = 3
	}
}

sd_is_independent_valid_target = {
	is_alive = yes
	is_landed = yes
	is_independent_ruler = yes
	is_adult = yes
	NOT = { this = scope:actor }
	NOT = { is_vassal_of = scope:actor }
}

sd_is_bound_tributary = {
	exists = var:sd_tributary_overlord
	var:sd_tributary_overlord = {
		is_alive = yes
	}
}

# this = potential overlord actor context uses scope:actor / scope:recipient
sd_can_offer_vassalage = {
	scope:actor = {
		sd_is_supreme_diplomat = yes
		sd_module_vassalage_on = yes
		sd_ai_may_use_diplomat = yes
		is_landed = yes
		is_independent_ruler = yes
		NOT = { has_character_flag = sd_vassalage_cd }
	}
	scope:recipient = {
		sd_is_independent_valid_target = yes
	}
	# Rank: prefer > at Low; ≥ at Mid/High
	trigger_if = {
		limit = { scope:actor = { sd_intensity_is_low = yes } }
		scope:actor.primary_title.tier > scope:recipient.primary_title.tier
	}
	trigger_else = {
		scope:actor.primary_title.tier >= scope:recipient.primary_title.tier
	}
}

sd_can_invite_realm = {
	scope:actor = {
		sd_is_supreme_diplomat = yes
		sd_module_realm_invite_on = yes
		sd_ai_may_use_diplomat = yes
		is_landed = yes
		is_independent_ruler = yes
		NOT = { sd_intensity_is_low = yes }
		NOT = { has_character_flag = sd_realm_invite_cd }
	}
	scope:recipient = {
		sd_is_independent_valid_target = yes
	}
	scope:actor.primary_title.tier > scope:recipient.primary_title.tier
	# Small realm gate by intensity (sub_realm_size = counties under them)
	trigger_if = {
		limit = { scope:actor = { sd_intensity_is_mid = yes } }
		scope:recipient = { sub_realm_size <= 12 }
	}
	trigger_else = {
		scope:recipient = { sub_realm_size <= 20 }
	}
}

# Adopt Liege Government — any scriptable liege gov (admin/feudal/clan/tribal/…).
# Hard-block unplayable / office / camp governments for actor and recipients.
sd_gov_is_hard_blocked_for_adopt = {
	OR = {
		government_has_flag = government_is_theocracy
		government_has_flag = government_is_ecclesiastical
		government_has_flag = government_is_republic
		government_has_flag = government_is_landless_adventurer
		government_has_flag = government_is_holy_order
		government_has_flag = government_is_mercenary
		government_has_flag = government_is_herder
		has_government = monastic_holy_order_government
	}
}

# Actor may offer Adopt Liege when their government is one we can mirror.
sd_actor_gov_is_adoptable = {
	NOT = { sd_gov_is_hard_blocked_for_adopt = yes }
}

# Keep old name as alias so any leftover refs stay valid
sd_actor_gov_is_feudal_or_clan = {
	sd_actor_gov_is_adoptable = yes
}

# this = potential convert; compare against top_liege (works in hub + iterators
# without requiring scope:actor). Lag-safe eligibility only — no world scans.
sd_eligible_for_liege_gov_adopt = {
	is_alive = yes
	is_landed = yes
	is_ai = yes
	is_adult = yes
	exists = top_liege
	NOT = { this = top_liege }
	NOT = { has_same_government = top_liege }
	NOT = { sd_gov_is_hard_blocked_for_adopt = yes }
	NOT = { holds_religious_office_trigger = yes }
	# Liege must have a government we can script a convert toward
	top_liege = { sd_actor_gov_is_adoptable = yes }
}

sd_can_adopt_liege_government = {
	scope:actor = {
		sd_is_supreme_diplomat = yes
		sd_ai_may_use_diplomat = yes
		is_landed = yes
		sd_actor_gov_is_adoptable = yes
		NOT = { has_character_flag = sd_adopt_liege_gov_cd }
	}
	scope:recipient = {
		# Anyone in my realm (direct or below), not only direct vassals
		top_liege = scope:actor
		sd_eligible_for_liege_gov_adopt = yes
	}
}

sd_actor_has_mismatched_realm_gov = {
	any_vassal_or_below = {
		sd_eligible_for_liege_gov_adopt = yes
	}
}
